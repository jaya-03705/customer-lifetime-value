    <script>
        let rawData = [];
        let processedData = [];
        let clvHistogramChart = null;
        let behaviorScatterChart = null;

        // Sample Data Generator matching standard behavioral/regression structure
        document.getElementById('loadSampleBtn').addEventListener('click', () => {
            rawData = [];
            const sampleIds = ['CUST-1001', 'CUST-1002', 'CUST-1003', 'CUST-1004', 'CUST-1005', 'CUST-1006', 'CUST-1007', 'CUST-1008', 'CUST-1009', 'CUST-1010'];
            for (let i = 0; i < 200; i++) {
                rawData.push({
                    CustomerID: i < 10 ? sampleIds[i] : `CUST-${1011 + i}`,
                    Recency: Math.floor(Math.random() * 365),
                    Frequency: Math.floor(Math.random() * 25) + 1,
                    Monetary: Math.floor(Math.random() * 1500) + 50
                });
            }
            runClvPipeline();
        });

        // File Upload Listener for CSV dataset
        document.getElementById('csvFileInput').addEventListener('change', (e) => {
            const file = e.target.files[0];
            if (!file) return;
            Papa.parse(file, {
                header: true,
                dynamicTyping: true,
                skipEmptyLines: true,
                complete: (results) => {
                    rawData = results.data;
                    runClvPipeline();
                }
            });
        });

        document.getElementById('recalculateBtn').addEventListener('click', () => {
            if (rawData.length > 0) runClvPipeline();
        });

        function guessColumn(row, keywords) {
            const keys = Object.keys(row);
            for (let kw of keywords) {
                const found = keys.find(k => k.toLowerCase().includes(kw));
                if (found) return found;
            }
            return keys[0];
        }

        function runClvPipeline() {
            if (rawData.length === 0) return;

            const sample = rawData[0];
            const colId = guessColumn(sample, ['id', 'customer', 'cust']);
            const colRec = guessColumn(sample, ['recency', 'recent', 'days']);
            const colFreq = guessColumn(sample, ['frequency', 'freq', 'count', 'transactions']);
            const colMon = guessColumn(sample, ['monetary', 'revenue', 'spend', 'amount', 'value']);

            const profitMargin = parseFloat(document.getElementById('profitMargin').value) / 100 || 0.2;
            const discountRate = parseFloat(document.getElementById('discountRate').value) / 100 || 0.1;
            const horizon = parseFloat(document.getElementById('timeHorizon').value) || 3;

            // Simple Linear Regression calculation for Monetary vs Frequency
            let n = rawData.length;
            let sumF = 0, sumM = 0, sumFM = 0, sumFF = 0;

            rawData.forEach(row => {
                let f = parseFloat(row[colFreq]) || 1;
                let m = parseFloat(row[colMon]) || 100;
                sumF += f;
                sumM += m;
                sumFM += f * m;
                sumFF += f * f;
            });

            let slope = (n * sumFM - sumF * sumM) / (n * sumFF - sumF * sumF || 1);
            let intercept = (sumM - slope * sumF) / n;

            // R-squared calculation
            let meanM = sumM / n;
            let ssTot = 0, ssRes = 0;
            rawData.forEach(row => {
                let f = parseFloat(row[colFreq]) || 1;
                let m = parseFloat(row[colMon]) || 100;
                let predM = intercept + slope * f;
                ssTot += Math.pow(m - meanM, 2);
                ssRes += Math.pow(m - predM, 2);
            });
            let rSquared = ssTot === 0 ? 0 : Math.max(0, 1 - (ssRes / ssTot));

            // Process individual CLV
            processedData = rawData.map(row => {
                let cid = row[colId] || 'Unknown';
                let rec = parseFloat(row[colRec]) || 30;
                let freq = parseFloat(row[colFreq]) || 1;
                let mon = parseFloat(row[colMon]) || 100;

                // Behavioral predictive adjustment
                let predictedAnnualRevenue = (intercept + slope * freq) * (freq > 0 ? freq : 1);
                let clv = 0;
                for (let t = 1; t <= horizon; t++) {
                    clv += (predictedAnnualRevenue * profitMargin) / Math.pow(1 + discountRate, t);
                }

                let segment = clv > 2000 ? 'VIP' : clv > 800 ? 'Loyal' : clv > 300 ? 'Potential' : 'At Risk';

                return { cid, rec, freq, mon, clv: Math.max(10, clv), segment };
            });

            updateKpis(rSquared);
            renderTable(processedData);
            renderCharts(processedData);
        }

        function updateKpis(rSquared) {
            const totalCust = processedData.length;
            const totalClvSum = processedData.reduce((acc, curr) => acc + curr.clv, 0);
            const avgClv = totalCust > 0 ? totalClvSum / totalCust : 0;

            document.getElementById('kpiTotalCustomers').innerText = totalCust.toLocaleString();
            document.getElementById('kpiAvgClv').innerText = '$' + avgClv.toFixed(2);
            document.getElementById('kpiTotalValue').innerText = '$' + totalClvSum.toLocaleString(undefined, {maximumFractionDigits: 0});
            document.getElementById('kpiRegressionFit').innerText = rSquared.toFixed(3);
        }

        function renderTable(data) {
            const tbody = document.getElementById('customerTableBody');
            tbody.innerHTML = '';
            
            data.slice(0, 50).forEach(row => {
                const tr = document.createElement('tr');
                tr.className = 'hover:bg-slate-50 transition';
                tr.innerHTML = `
                    <td class="p-4 font-medium text-slate-900">${row.cid}</td>
                    <td class="p-4 text-slate-600">${row.rec}</td>
                    <td class="p-4 text-slate-600">${row.freq}</td>
                    <td class="p-4 text-slate-600">$${row.mon.toFixed(2)}</td>
                    <td class="p-4 font-semibold text-indigo-600">$${row.clv.toFixed(2)}</td>
                    <td class="p-4"><span class="px-2.5 py-1 text-xs font-semibold rounded-full ${getBadgeStyle(row.segment)}">${row.segment}</span></td>
                `;
                tbody.appendChild(tr);
            });
        }

        function getBadgeStyle(segment) {
            switch(segment) {
                case 'VIP': return 'bg-purple-100 text-purple-800';
                case 'Loyal': return 'bg-emerald-100 text-emerald-800';
                case 'Potential': return 'bg-blue-100 text-blue-800';
                default: return 'bg-rose-100 text-rose-800';
            }
        }

        function renderCharts(data) {
            const ctxHist = document.getElementById('clvHistogramChart').getContext('2d');
            const ctxScatter = document.getElementById('behaviorScatterChart').getContext('2d');

            if (clvHistogramChart) clvHistogramChart.destroy();
            if (behaviorScatterChart) behaviorScatterChart.destroy();

            // Histogram Bins
            const bins = { '0-500': 0, '500-1500': 0, '1500-3000': 0, '3000-5000': 0, '5000+': 0 };
            data.forEach(d => {
                if (d.clv <= 500) bins['0-500']++;
                else if (d.clv <= 1500) bins['500-1500']++;
                else if (d.clv <= 3000) bins['1500-3000']++;
                else if (d.clv <= 5000) bins['3000-5000']++;
                else bins['5000+']++;
            });

            clvHistogramChart = new Chart(ctxHist, {
                type: 'bar',
                data: {
                    labels: Object.keys(bins),
                    datasets: [{
                        label: 'Customer Count',
                        data: Object.values(bins),
                        backgroundColor: '#6366f1',
                        borderRadius: 6
                    }]
                },
                options: { responsive: true, maintainAspectRatio: false }
            });

            behaviorScatterChart = new Chart(ctxScatter, {
                type: 'scatter',
                data: {
                    datasets: [{
                        label: 'Frequency vs Monetary',
                        data: data.map(d => ({ x: d.freq, y: d.mon })),
                        backgroundColor: '#10b981'
                    }]
                },
                options: { responsive: true, maintainAspectRatio: false }
            });
        }

        // Search Filter
        document.getElementById('searchInput').addEventListener('input', (e) => {
            const term = e.target.value.toLowerCase();
            const filtered = processedData.filter(d => String(d.cid).toLowerCase().includes(term));
            renderTable(filtered);
        });

        // CSV Export
        document.getElementById('exportCsvBtn').addEventListener('click', () => {
            if (processedData.length === 0) return;
            let csvContent = "data:text/csv;charset=utf-8,CustomerID,Recency,Frequency,Monetary,PredictedCLV,Segment\n";
            processedData.forEach(row => {
                csvContent += `${row.cid},${row.rec},${row.freq},${row.mon},${row.clv.toFixed(2)},${row.segment}\n`;
            });
            const encodedUri = encodeURI(csvContent);
            const link = document.createElement("a");
            link.setAttribute("href", encodedUri);
            link.setAttribute("download", "clv_regression_predictions.csv");
            document.body.appendChild(link);
            link.click();
            document.body.removeChild(link);
        });
    </script>
